compare this yml file to current yml. name: "Windows Build"

on:
  push:
    branches: [ "**" ]
  pull_request:
    branches: [ "**" ]
  workflow_dispatch:

env:
  JDK_VERSION: 21
  JDK_DISTRO: 'temurin'

jobs:
  build_windows:
    name: 'Windows build'
    runs-on: windows-latest

    steps:
      - name: "Checkout sources"
        uses: actions/checkout@v7

      - name: Set up JDK ${{ env.JDK_VERSION }} ${{ env.JDK_DISTRO }}
        uses: actions/setup-java@v5
        with:
          java-version: ${{ env.JDK_VERSION }}
          distribution: ${{ env.JDK_DISTRO }}

      - name: Set up Gradle caching
        uses: gradle/actions/setup-gradle@v4

      - name: "Build MSI"
        run: .\gradlew.bat createMsi -x checkstyleMain -x checkstyleTest

      - name: "Build self-contained archive for Windows"
        run: .\gradlew.bat createWindowsPortableZip -x checkstyleMain -x checkstyleTest

      - name: "Upload Windows MSI"
        uses: actions/upload-artifact@v7
        if: success()
        with:
          path: build/dist/*.msi
          name: logisim-evolution-windows-msi

      - name: "Upload Windows Portable ZIP"
        uses: actions/upload-artifact@v7
        if: success()
        with:
          path: build/dist/*.zip
          name: logisim-evolution-windows-portable-zip
